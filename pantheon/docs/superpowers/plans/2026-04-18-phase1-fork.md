# Phase 1 — Open WebUI Fork & Sanctuary Integration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Get a running local Pantheon instance — Open WebUI serving the Sanctuary selector UI, backed by a pantheon-core sidecar that enforces harness-based prompt isolation and real-time vault logging.

**Architecture:** Open WebUI is pulled as a git subtree into `open-webui/`. `pantheon-core` runs as a FastAPI sidecar service. Open WebUI's backend gets two minimal modifications: a middleware that intercepts chat requests and injects the harness identity as the system prompt, and proxy routes that forward `/api/pantheon/*` to pantheon-core. The Sanctuary selector replaces Open WebUI's sidebar. All vault writes land directly on the filesystem via a Docker volume mount.

**Tech Stack:** Python 3.12, FastAPI, uvicorn, PyYAML, httpx, pytest, SvelteKit (Open WebUI frontend), Docker Compose.

**Spec:** `docs/superpowers/specs/2026-04-18-phase1-fork-design.md`

```python
# pantheon-core/sanctuary/config.py
from __future__ import annotations

import os
from dataclasses import dataclass, field
from pathlib import Path

import yaml

SANCTUARIES_DIR = Path(
    os.environ.get(
        "PANTHEON_SANCTUARIES_DIR",
        Path(__file__).resolve().parents[2]
        / "Athenaeum/Codex-Pantheon/harnesses/sanctuaries",
    )
)


@dataclass
class VaultLogging:
    enabled: bool
    path: str
    format: str = "markdown"
    filename: str = "timestamp"


@dataclass
class SanctuaryUI:
    accent_color: str = "#6b7280"
    icon: str = "⚡"
    description: str = ""


@dataclass
class SanctuaryConfig:
    id: str
    name: str
    god: str
    harness: str
    model: str
    context_window: int
    vault_logging: VaultLogging
    ui: SanctuaryUI
    studio: str | None = None


def _parse(path: Path) -> SanctuaryConfig:
    data = yaml.safe_load(path.read_text())
    vl = data.get("vault_logging", {})
    ui = data.get("ui", {})
    return SanctuaryConfig(
        id=path.stem,
        name=data["name"],
        god=data["god"],
        studio=data.get("studio"),
        harness=data["harness"],
        model=data.get("model", "gemma4"),
        context_window=data.get("context_window", 8192),
        vault_logging=VaultLogging(
            enabled=vl.get("enabled", False),
            path=vl.get("path", ""),
            format=vl.get("format", "markdown"),
            filename=vl.get("filename", "timestamp"),
        ),
        ui=SanctuaryUI(
            accent_color=ui.get("accent_color", "#6b7280"),
            icon=ui.get("icon", "⚡"),
            description=ui.get("description", ""),
        ),
    )


def load_sanctuary(sanctuary_id: str) -> SanctuaryConfig | None:
    path = SANCTUARIES_DIR / f"{sanctuary_id}.yaml"
    if not path.exists():
        return None
    return _parse(path)


def load_all_sanctuaries() -> list[SanctuaryConfig]:
    if not SANCTUARIES_DIR.exists():
        return []
    return [_parse(p) for p in sorted(SANCTUARIES_DIR.glob("*.yaml"))]
```

# pantheon-core/vault/writer.py
from __future__ import annotations

from datetime import datetime, timezone
from pathlib import Path

from sanctuary.config import SanctuaryConfig


class VaultWriter:
    def __init__(self, athenaeum_root: str) -> None:
        self._root = Path(athenaeum_root)
        self._sessions: dict[str, Path | None] = {}

    def append_turn(
        self,
        session_id: str,
        sanctuary: SanctuaryConfig,
        role: str,
        content: str,
    ) -> None:
        if not sanctuary.vault_logging.enabled:
            return
        path = self._get_or_create(session_id, sanctuary)
        role_label = "User" if role == "user" else sanctuary.god
        turn = f"\n[{role_label}]: {content}\n"
        with path.open("a", encoding="utf-8") as f:
            f.write(turn)

    def _get_or_create(self, session_id: str, sanctuary: SanctuaryConfig) -> Path:
        if session_id not in self._sessions:
            self._sessions[session_id] = self._create_session_file(sanctuary)
        return self._sessions[session_id]

    def _create_session_file(self, sanctuary: SanctuaryConfig) -> Path:
        timestamp = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H-%M-%S")
        filename = f"{timestamp}.md"
        rel_path = sanctuary.vault_logging.path.strip("/")
        path = self._root / rel_path / filename
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(self._build_header(sanctuary, timestamp), encoding="utf-8")
        return path

    def _build_header(self, sanctuary: SanctuaryConfig, timestamp: str) -> str:
        lines = [
            "---",
            f"sanctuary: {sanctuary.name}",
            f"god: {sanctuary.god}",
        ]
        if sanctuary.studio:
            lines.append(f"studio: {sanctuary.studio}")
        lines += [f"timestamp: {timestamp}", "---\n"]
        return "\n".join(lines)
```

# pantheon-core/api.py
from __future__ import annotations

import os
from pathlib import Path

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel

from harness.loader import load_harness
from sanctuary.config import load_all_sanctuaries, load_sanctuary
from vault.writer import VaultWriter

app = FastAPI(title="Pantheon Core")

ATHENAEUM_ROOT = os.environ.get("ATHENAEUM_ROOT", "/athenaeum")
_writer = VaultWriter(ATHENAEUM_ROOT)


@app.get("/sanctuaries")
def get_sanctuaries() -> dict:
    sanctuaries = load_all_sanctuaries()
    grouped: dict[str, list] = {}
    for s in sanctuaries:
        entry = {
            "id": s.id,
            "name": s.name,
            "god": s.god,
            "studio": s.studio,
            "model": s.model,
            "ui": {
                "accent_color": s.ui.accent_color,
                "icon": s.ui.icon,
                "description": s.ui.description,
            },
        }
        grouped.setdefault(s.god, []).append(entry)
    return grouped


@app.get("/sanctuary/{sanctuary_id}/prompt")
def get_prompt(sanctuary_id: str) -> dict:
    sanctuary = load_sanctuary(sanctuary_id)
    if sanctuary is None:
        raise HTTPException(status_code=404, detail=f"Sanctuary not found: {sanctuary_id}")
    harness = load_harness(sanctuary.harness)
    identity = harness.get("identity")
    if not identity:
        raise HTTPException(
            status_code=500, detail=f"Harness has no identity field: {sanctuary.harness}"
        )
    return {"prompt": identity.strip(), "god": sanctuary.god, "sanctuary": sanctuary.name}


class LogEntry(BaseModel):
    session_id: str
    role: str
    content: str


@app.post("/sanctuary/{sanctuary_id}/log")
def log_turn(sanctuary_id: str, entry: LogEntry) -> dict:
    sanctuary = load_sanctuary(sanctuary_id)
    if sanctuary is None:
        raise HTTPException(status_code=404, detail=f"Sanctuary not found: {sanctuary_id}")
    _writer.append_turn(entry.session_id, sanctuary, entry.role, entry.content)
    return {"status": "ok"}
```

# open-webui/backend/open_webui/pantheon_middleware.py
from __future__ import annotations

import json
import os

import httpx
from starlette.middleware.base import BaseHTTPMiddleware
from starlette.requests import Request
from starlette.responses import Response

PANTHEON_CORE_URL = os.environ.get("PANTHEON_CORE_URL", "http://pantheon-core:8000")

CHAT_PATHS = ("/api/chat/completions", "/ollama/api/chat", "/openai/chat/completions")


class PantheonMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next) -> Response:
        if request.method == "POST" and any(p in request.url.path for p in CHAT_PATHS):
            sanctuary_id = request.headers.get("X-Pantheon-Sanctuary-Id")
            if sanctuary_id:
                request = await self._inject_prompt(request, sanctuary_id)
        return await call_next(request)

    async def _inject_prompt(self, request: Request, sanctuary_id: str) -> Request:
        body = await request.body()
        try:
            data = json.loads(body)
        except json.JSONDecodeError:
            return request

        try:
            async with httpx.AsyncClient() as client:
                resp = await client.get(
                    f"{PANTHEON_CORE_URL}/sanctuary/{sanctuary_id}/prompt",
                    timeout=5.0,
                )
            if resp.status_code != 200:
                return request
            prompt = resp.json()["prompt"]
        except Exception:
            return request

        messages = [m for m in data.get("messages", []) if m.get("role") != "system"]
        messages.insert(0, {"role": "system", "content": prompt})
        data["messages"] = messages

        modified = json.dumps(data).encode()

        async def receive():
            return {"type": "http.request", "body": modified, "more_body": False}

        return Request(request.scope, receive)
```

// open-webui/src/lib/stores/pantheon.ts
import { writable, get } from 'svelte/store';

export interface SanctuaryUI {
  accent_color: string;
  icon: string;
  description: string;
}

export interface SanctuaryEntry {
  id: string;
  name: string;
  god: string;
  studio: string | null;
  model: string;
  ui: SanctuaryUI;
}

export interface SanctuaryGroups {
  [god: string]: SanctuaryEntry[];
}

export const sanctuaries = writable<SanctuaryGroups>({});
export const activeSanctuaryId = writable<string | null>(null);
export const activeSanctuaryPrompt = writable<string | null>(null);

export async function loadSanctuaries(): Promise<void> {
  try {
    const resp = await fetch('/api/pantheon/sanctuaries');
    if (resp.ok) {
      sanctuaries.set(await resp.json());
    }
  } catch (e) {
    console.error('Failed to load sanctuaries', e);
  }
}

export async function selectSanctuary(id: string): Promise<void> {
  activeSanctuaryId.set(id);
  activeSanctuaryPrompt.set(null);
}

export function clearSanctuary(): void {
  activeSanctuaryId.set(null);
  activeSanctuaryPrompt.set(null);
}

export async function logTurn(role: string, content: string): Promise<void> {
  const id = get(activeSanctuaryId);
  if (!id) return;
  // Generate session ID once per page load — stored in module scope
  await fetch(`/api/pantheon/sanctuary/${id}/log`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ session_id: _sessionId, role, content }),
  });
}

// One session ID per page load
const _sessionId = crypto.randomUUID();
```
